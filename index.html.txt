<script>
    // Danh sách từ vựng mẫu để luyện tập
    const wordList = [
        { cn: "你好", pinyin: "nǐ hǎo" },
        { cn: "谢谢", pinyin: "xièxie" },
        { cn: "再见", pinyin: "zàijiàn" },
        { cn: "中国", pinyin: "Zhōngguó" },
        { cn: "我想学汉语", pinyin: "wǒ xiǎng xué Hànyǔ" },
        { cn: "今天天气很好", pinyin: "jīntiān tiānqì hěn hǎo" },
        { cn: "你是哪国人", pinyin: "nǐ shì nǎ guó rén" }
    ];

    let currentWord = wordList[0];
    
    // Kiểm tra hỗ trợ trình duyệt
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
        alert("Trình duyệt của bạn không hỗ trợ nhận dạng giọng nói. Hãy dùng Google Chrome!");
    }

    const recognition = new SpeechRecognition();
    recognition.lang = 'zh-CN'; // Cấu hình nhận diện tiếng Trung Quốc (Phổ thông)
    recognition.interimResults = false;
    recognition.maxAlternatives = 1;

    const recordBtn = document.getElementById('record-btn');
    const nextBtn = document.getElementById('next-btn');
    const targetCnDisplay = document.getElementById('target-cn');
    const targetPinyinDisplay = document.getElementById('target-pinyin');
    const resultSection = document.getElementById('result-section');
    const userTranscript = document.getElementById('user-transcript');
    const scoreDisplay = document.getElementById('score-display');
    const feedbackMsg = document.getElementById('feedback-msg');

    let isRecording = false;

    // Xử lý sự kiện Ghi âm
    recordBtn.addEventListener('click', () => {
        if (!isRecording) {
            recognition.start();
        } else {
            recognition.stop();
        }
    });

    recognition.onstart = () => {
        isRecording = true;
        recordBtn.textContent = "Đang lắng nghe... Hãy nói đi!";
        recordBtn.classList.add('recording');
        resultSection.style.display = 'none';
    };

    recognition.onend = () => {
        isRecording = false;
        recordBtn.textContent = "Nhấn để nói";
        recordBtn.classList.remove('recording');
    };

    recognition.onresult = (event) => {
        const resultText = event.results[0][0].transcript;
        // Loại bỏ dấu câu nếu có để so sánh chính xác hơn
        const cleanResult = resultText.replace(/[.,\/#!$%\^&\*;:{}=\-_`~()？。，！]/g,"");
        const cleanTarget = currentWord.cn.replace(/[.,\/#!$%\^&\*;:{}=\-_`~()？。，！]/g,"");

        userTranscript.textContent = resultText;
        
        // Tính toán điểm số dựa trên độ tương đồng chuỗi
        const score = Math.round(calculateSimilarity(cleanTarget, cleanResult) * 100);
        
        displayResult(score);
    };

    recognition.onerror = (event) => {
        alert("Đã xảy ra lỗi nhận diện: " + event.error);
    };

    // Thuật toán Dice's Coefficient để chấm điểm so sánh 2 chuỗi ký tự Trung Quốc
    function calculateSimilarity(str1, str2) {
        if (str1 === str2) return 1.0;
        if (str1.length < 2 || str2.length < 2) {
            // Nếu từ quá ngắn, so sánh ký tự trực tiếp
            let matches = 0;
            for(let char of str1) {
                if(str2.includes(char)) matches++;
            }
            return matches / Math.max(str1.length, str2.length);
        }

        const getBigrams = str => {
            const bigrams = new Set();
            for (let i = 0; i < str.length - 1; i++) {
                bigrams.add(str.substring(i, i + 2));
            }
            return bigrams;
        };

        const bigrams1 = getBigrams(str1);
        const bigrams2 = getBigrams(str2);
        
        let intersection = 0;
        for (let bigram of bigrams1) {
            if (bigrams2.has(bigram)) intersection++;
        }

        return (2.0 * intersection) / (bigrams1.size + bigrams2.size);
    }

    // Hiển thị kết quả & Đánh giá
    function displayResult(score) {
        resultSection.style.display = 'block';
        scoreDisplay.textContent = score + "%";
        
        // Reset class màu sắc
        scoreDisplay.className = "score";
        
        if (score >= 80) {
            scoreDisplay.classList.add('good');
            feedbackMsg.textContent = "Quá tuyệt vời! Phát âm rất chuẩn.";
            feedbackMsg.className = "good";
        } else if (score >= 50) {
            scoreDisplay.classList.add('average');
            feedbackMsg.textContent = "Khá tốt, nhưng cần chú ý hơn thanh điệu nhé.";
            feedbackMsg.className = "average";
        } else {
            scoreDisplay.classList.add('bad');
            feedbackMsg.textContent = "Chưa chính xác lắm. Thử lại lần nữa nào!";
            feedbackMsg.className = "bad";
        }
    }

    // Đổi từ ngẫu nhiên
    nextBtn.addEventListener('click', () => {
        const randomIndex = Math.floor(Math.random() * wordList.length);
        currentWord = wordList[randomIndex];
        targetCnDisplay.textContent = currentWord.cn;
        targetPinyinDisplay.textContent = currentWord.pinyin;
        resultSection.style.display = 'none';
    });
</script>
